#Practical 1: Find cofactors, determinant, adjoint and inverse of a matrix.
import numpy as np

NR=int(input('Enter no. of rows:'))
NC=int(input('Enter no. of columns:'))

print('Enter the entries in a single line(separated by space ):')
Entries=list(map(int,input().split()))

#For printing the matrix
A=np.array(Entries).reshape(NR,NC)
print('Matrix A is as follows:',"\n",A)

#For finding the inverse of the matrix
A_inverse=np.linalg.inv(A)
print('Inverse of A is','\n',A_inverse)

#For finding the transpose of the matrix
Transpose_A=np.transpose(A_inverse)
print('Transpose of A is','\n',Transpose_A)

#For finding the determinant of the matrix
determinant_A=np.linalg.det(A)
print('Determinant of A is','\n',determinant_A)

#For finding the cofactors of the matrix
cofactors_A=np.dot(Transpose_A,determinant_A)
print('the cofactor of a matrix is:','\n',cofactors_A)

#For finding the Adjoint of the matrix
Adjoint_A=np.transpose(cofactors_A)
print('The adjoint of a matrix is:','\n',Adjoint_A)

#Practical 5: Verify the linear dependence of vectors.Generate a linear combination of given vectors of R^n/ matrices of the same size.
import numpy as np

# Input: Number of vectors
m = int(input("Enter the number of vectors: "))
n = int(input("Enter the dimension of the vectors (n for R^n): "))

# Input: Vectors
vectors = []
print("Enter the vectors, each as space-separated values:")
for i in range(m):
    vectors.append(list(map(float, input(f"Vector {i+1}: ").split())))

vectors = np.array(vectors)

# Step 1: Check linear dependence
rank = np.linalg.matrix_rank(vectors)
print(f"\nRank of the matrix formed by the vectors: {rank}")
if rank < m:
    print("The vectors are linearly dependent.")
else:
    print("The vectors are linearly independent.")

# Step 2: Generate a linear combination
print("\nEnter coefficients for linear combination (space-separated):")
coefficients = np.array(list(map(float, input().split())))

# Ensure the number of coefficients matches the number of vectors
if len(coefficients) != m:
    print("Error: Number of coefficients must match the number of vectors.")
else:
    linear_combination = np.dot(coefficients, vectors)
    print("\nLinear combination of the given vectors:")
    print(linear_combination)





#Practical 6:Check the diagonalizable property of matrices and find the corresponding eigenvalue and verify the Cayley Hamilton theorem.

import numpy as np

# Input: Square matrix
n = int(input("Enter the size of the square matrix (n x n): "))
print("Enter the elements of the matrix row-wise:")
matrix = np.array([list(map(float, input().split())) for _ in range(n)])

# Step 1: Compute eigenvalues and eigenvectors
eigenvalues, eigenvectors = np.linalg.eig(matrix)

print("\nEigenvalues:")
print(eigenvalues)

print("\nEigenvectors:")
print(eigenvectors)

# Step 2: Check if the matrix is diagonalizable
if np.linalg.matrix_rank(eigenvectors) == n:
    print("\nThe matrix is diagonalizable.")
else:
    print("\nThe matrix is not diagonalizable.")

# Step 3: Verify the Cayley-Hamilton theorem
# Compute the characteristic polynomial coefficients
char_poly = np.poly(matrix)  # Coefficients of the characteristic polynomial
print("\nCharacteristic polynomial coefficients (highest to lowest degree):")
print(char_poly)

# Reconstruct the polynomial expression in terms of the matrix
char_poly_matrix = np.zeros_like(matrix)
for i, coeff in enumerate(char_poly):
    char_poly_matrix += coeff * np.linalg.matrix_power(matrix, n - i - 1)

# Add the constant term
char_poly_matrix += char_poly[-1] * np.eye(n)

print("\nVerification of Cayley-Hamilton theorem (Should be a zero matrix):")
print(char_poly_matrix)

if np.allclose(char_poly_matrix, np.zeros_like(matrix)):
    print("The Cayley-Hamilton theorem is verified.")
else:
    print("The Cayley-Hamilton theorem is not verified.")






#Practical 7: Compute Gradient of a scalar field, Divergence and Curl of a vector field. 
import sympy as sp

# Define symbols (coordinates)
x, y, z = sp.symbols('x y z')

# Example Scalar Field f(x, y, z)
f = x**2 + y**2 + z**2

# Example Vector Field A(x, y, z)
A_x = x * y
A_y = y * z
A_z = z * x
A = sp.Matrix([A_x, A_y, A_z])

# 1. Compute the Gradient of the scalar field f
gradient_f = sp.Matrix([sp.diff(f, var) for var in (x, y, z)])
print("Gradient of f(x, y, z):")
sp.pprint(gradient_f)

# 2. Compute the Divergence of the vector field A
divergence_A = sp.diff(A_x, x) + sp.diff(A_y, y) + sp.diff(A_z, z)
print("\nDivergence of A(x, y, z):")
sp.pprint(divergence_A)

# 3. Compute the Curl of the vector field A
curl_A = sp.Matrix([
    sp.diff(A_z, y) - sp.diff(A_y, z),  # i-component
    sp.diff(A_x, z) - sp.diff(A_z, x),  # j-component
    sp.diff(A_y, x) - sp.diff(A_x, y)   # k-component
])
print("\nCurl of A(x, y, z):")
sp.pprint(curl_A)
